{{- /* Page-bundle raster images are resized (WebP, srcset) so posts don't load the multi-MB original. The image links to a full-size copy capped at 3000px on the long edge (never upscaled, re-encoded so EXIF/GPS is dropped); the raw original is not referenced, so Hugo doesn't publish it. Paths are resolved to permalinks so they also work where the post body is rendered on another page. Remote images, static files, SVG and GIF render as a plain img. */ -}} {{- $src := .Destination -}} {{- /* The first image in a body is the top image on a post page, i.e. the LCP element: load it eagerly with high priority; the rest stay lazy (WEBS-29). */ -}} {{- $loading := cond (eq .Ordinal 0) `fetchpriority="high"` `loading="lazy"` -}} {{- $res := "" -}} {{- if and (not (urls.Parse $src).Scheme) (not (hasPrefix $src "/")) -}} {{- $res = .PageInner.Resources.Get $src -}} {{- end -}} {{- if and $res (eq $res.MediaType.MainType "image") (not (in (slice "svg" "gif") $res.MediaType.SubType)) -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $w := slice 480 800 1200 1600 -}} {{- if lt $w $res.Width -}} {{- $img := $res.Resize (printf "%dx webp q80" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $w) -}} {{- if eq $w 800 -}}{{- $fallback = $img -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $fallback -}} {{- $fallback = $res.Resize (printf "%dx webp q80" $res.Width) -}} {{- $srcset = $srcset | append (printf "%s %dw" $fallback.RelPermalink $res.Width) -}} {{- end -}} {{- $full := $res.Fit "3000x3000 q85" -}} {{- else -}} {{- with $res -}}{{- $src = .RelPermalink -}}{{- end -}} {{- end -}}
